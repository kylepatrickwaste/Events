/**********************************************************************************
  PART 1 of 2  --  SCHEMA-FIT SCRIPT
  Make the simplified "Events" database able to hold EVERY column from the
  complex "RouteEvents" (RouteEventsDB.sql) source, per the request to preserve
  all information.

  Strategy chosen: ADD COLUMNS to fit the orphan source columns onto the existing
  Events tables, and CREATE new tables only where the source data is genuinely
  relational and has no home in the 10-table design.

  Ignored source tables (per rule -- CUBE suffix, or z / z_ prefix):
        RouteEventsCube , z_RE_IMPORT_CHECK , z_RE_TEMP

  ------------------------------------------------------------------------------
  COVERAGE MATRIX  (44 in-scope source tables)
  ------------------------------------------------------------------------------
  A. FITS AN EXISTING Events TABLE (with ALTER ADD for orphan columns)
     Districts                 -> Districts     (+ audit, ToolboxDist, IsRmoPreselected)
     EventTypes                -> EventTypes    (+ audit)
     EventSources              -> EventSources  (+ audit)
     FileImports               -> FileImports   (+ Active, audit)
     ServiceCodeXDistrict      -> ServiceCodes  (Name->Code; + PreferRMOAmount, ReplacedBy, audit)
     RouteEvents               -> RouteEvents   (+ DateCreated, CustomerNumber, EventUrl,
                                                  FinalActionXRouteEventId, RouteType,
                                                  EquipmentType, BodyType, Stop* parcels,
                                                  StopLatitude/Longitude, AssociationMatch)
     ActionXRouteEvent
       + ChargeXAction         -> EventActions  (charge-line grain; + ActionId, ChargeXActionId,
                                                  CloseReasonId/Details, batch ids, charge cust/acct,
                                                  BilledInfoLookUpOn, audit)
     AddressEditHistory
       + QuantityEditHistory   -> EventEditHistory (+ ActionId)
     ExcludedAccounts          -> AccountFlags  (Flag='excluded'; + CustomerNumber, audit)
     UserInfo                  -> AppUsers       (NtUsername->ActiveDirectoryName; clean fit)

  B. ABSORBED BY DENORMALIZATION (no table needed; catalog collapses to a string)
     Actions                   -> EventActions.ActionType
     CloseReasonXAction        -> EventActions.CloseReason
     ExcludedReasons           -> ExcludedAccountScopes.ExcludedReason (see new table)

  C. NEW TABLES REQUIRED -- relational event-domain data with no home
     ExcludedAccountScopes     <- ExcludedAccountXEventTypeXEventSource
                                  + ExcludedReasonsXExcludedAccounts (many per account,
                                    so cannot collapse into AccountFlags' unique key)
     ServiceCodeEventTypes     <- ServiceCodeXEventType
     EmailXRouteEvent          <- EmailXRouteEvent
     AddressLookupRequests     <- AddressLookupRequests
     AddressLookupResults      <- AddressLookupResults
     FeatureXDistrict          <- FeatureXDistrict
     FeatureEditHistory        <- FeatureEditHistory

  D. ANCILLARY / STANDALONE DATA -- not part of the event/charge domain model.
     These carry telematics, reporting rollups, or operational logs. They have no
     FK relationship to the redesigned tables. They are recreated verbatim in the
     Events DB (Section D) so no data is lost, but they are intentionally left OUT
     of the simplified model -- delete the ones you don't need.
        Telematics/report : HealthReport, HealthReportBadPingData, HealthReportDate,
                            CoachingData, SafetyData, ReversingData, MOABSessionData,
                            RouteEventsSummary, EventsClosedNotCharged, WVPilotVehicles,
                            RolloutSchedule, CurrExRate, CurrencyExchangeReportDate
        Operational logs  : FileImportsDupLog, RouteEventDupDismissLog,
                            RouteEventsDeDupeLog, RouteEventsExported,
                            RouteEventsImportCheck, RouteEventsPreChange
        One-off staging   : RMOKnoxvilleExcludedAccountXEventTypeXEvents_AaronFelts,
                            RMOKnoxvilleExemptAccountLoad_AaronFelts

  ==> ANSWER: Yes, all information CAN be stored in the Events database, but not in
      only 10 tables. The event/charge domain fits with column additions + 7 small
      relational tables (Sections B/C). The ancillary/log/staging data (Section D)
      is independent of the redesign and is preserved as carry-over tables.

  NOTE ON KEYS: source RouteEvents.Id / ActionXRouteEvent.RouteEventId are BIGINT;
      Events.RouteEvents.Id / EventActions.RouteEventId are INT. Preserving original
      ids (IDENTITY_INSERT) requires every migrated id to fit in INT
      (<= 2,147,483,647). If your ids exceed that, change those columns to BIGINT
      first (ALTER shown at the very bottom, commented out).
**********************************************************************************/

-------------------------------------------------- ExcludedAccountScopes
-- Preserves the two many-to-one exclusion junctions that cannot collapse into
-- AccountFlags (its UNIQUE (DistrictId,AccountNumber,Flag) allows only one row
-- per account/flag, but an account can have many type/source and reason rows).
CREATE TABLE dbo.ExcludedAccountScopes (
    Id            int IDENTITY(1,1) NOT NULL,
    AccountFlagId int               NOT NULL,   -- -> AccountFlags.Id
    EventTypeId   int               NULL,       -- -> EventTypes.Id
    EventSourceId int               NULL,       -- -> EventSources.Id
    ExcludedReason nvarchar(50)     NULL,       -- was ExcludedReasons.Name
    Notes         nvarchar(500)     NULL,
    Active        bit               NOT NULL CONSTRAINT DF_EAS_Active DEFAULT (1),
    CreatedBy     nvarchar(50)      NULL,
    DateCreated   datetimeoffset(7) NOT NULL CONSTRAINT DF_EAS_DateCreated DEFAULT (sysdatetimeoffset()),
    CONSTRAINT PK_ExcludedAccountScopes PRIMARY KEY CLUSTERED (Id)
);
ALTER TABLE dbo.ExcludedAccountScopes ADD CONSTRAINT FK_EAS_AccountFlags
    FOREIGN KEY (AccountFlagId) REFERENCES dbo.AccountFlags (Id);
ALTER TABLE dbo.ExcludedAccountScopes ADD CONSTRAINT FK_EAS_EventTypes
    FOREIGN KEY (EventTypeId) REFERENCES dbo.EventTypes (Id);
ALTER TABLE dbo.ExcludedAccountScopes ADD CONSTRAINT FK_EAS_EventSources
    FOREIGN KEY (EventSourceId) REFERENCES dbo.EventSources (Id);
GO

-------------------------------------------------- ServiceCodeEventTypes

    Id            int IDENTITY(1,1) NOT NULL,

    EventTypeId   int               NOT NULL,   -- -> EventTypes.Id
    Active        bit               NOT NULL CONSTRAINT DF_SCET_Active DEFAULT (1),
    CreatedBy     nvarchar(50)      NULL,
    DateCreated   datetimeoffset(7) NOT NULL CONSTRAINT DF_SCET_DateCreated DEFAULT (sysdatetimeoffset()),
    UpdatedBy     nvarchar(50)      NULL,
    DateUpdated   datetimeoffset(7) NULL,
    CONSTRAINT PK_ServiceCodeEventTypes PRIMARY KEY CLUSTERED (Id)
);
ALTER TABLE dbo.ServiceCodeEventTypes ADD CONSTRAINT FK_SCET_ServiceCodes
    FOREIGN KEY (ServiceCodeId) REFERENCES dbo.ServiceCodes (Id);
ALTER TABLE dbo.ServiceCodeEventTypes ADD CONSTRAINT FK_SCET_EventTypes
    FOREIGN KEY (EventTypeId) REFERENCES dbo.EventTypes (Id);
GO
